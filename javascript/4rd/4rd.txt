    <script>
        function showPrice() {
            var originPrice1 = Number(document.querySelector("#oPrice1").value);
            var originPrice2 = Number(document.querySelector("#oPrice2").value);
            var originPrice3 = Number(document.querySelector("#oPrice3").value);

            var originPrice = originPrice1 + originPrice2 + originPrice3;

            var world = originPrice;
            var result;
            var savedPrice;

            if (world > 50000) {
                result = "VIP";
                savedPrice = originPrice * 0.003;
            }
            else if (world >= 30000) {
                result = "GOLD";
                savedPrice = originPrice * 0.002;
            }
            else if (world >= 10000) {
                result = "SILVER";
                savedPrice = originPrice * 0.001;
            }

            document.querySelector("#showResult").innerHTML =
                "상품의 가격은 " + originPrice +
                "원이고, 등급은 " + result + "이며 " +
                savedPrice + "원의 적립금을 얻을 수 있습니다.";
        }
    </script>
